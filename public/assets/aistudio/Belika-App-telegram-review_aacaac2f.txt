  if (sizes === 'M–3XL') return ['M', 'L', 'XL', 'XXL', '3XL'];
  if (sizes === 'L–XXL') return ['L', 'XL', 'XXL'];
  if (sizes === 'Free size') return ['Free size'];
  return ['Unspecified'];
};

const DEFAULT_PRODUCTS: Product[] = PRODUCTS.map((product) => {
  const colors = product.imageLabels?.length ? product.imageLabels : ['Unspecified'];
  const variants = getInitialSizes(product.sizes).flatMap((size, sizeIndex) =>
    colors.map((color, colorIndex) => ({
      id: `${product.id}-v${sizeIndex}-${colorIndex}`,
      size,
      color,
      stock: 3,
    })),
  );
  return { ...product, variants, status: 'published' };
});

function ZoomablePhoto({
  source,
  resizeMode,
}: {
  source: any;
  resizeMode: 'contain' | 'cover';
}) {
  const scale = useRef(new Animated.Value(1)).current;
  const currentScale = useRef(1);
  const startScale = useRef(1);
  const pinchGesture = useMemo(
    () =>
      Gesture.Pinch()
        .runOnJS(true)
        .onBegin(() => {
          startScale.current = currentScale.current;
        })
        .onUpdate((event) => {
          const nextScale = Math.min(3, Math.max(1, startScale.current * event.scale));
          currentScale.current = nextScale;
          scale.setValue(nextScale);
        })
        .onEnd(() => {
          startScale.current = currentScale.current;
        }),
    [scale],
  );

  return (
    <GestureDetector gesture={pinchGesture}>
      <View style={styles.zoomTouchArea} collapsable={false}>
        <Animated.Image
          source={source}
          style={[styles.detailRealPhoto, { transform: [{ scale }] }]}
          resizeMode={resizeMode}
        />
      </View>
    </GestureDetector>
  );
}

    if (product.sizes === 'M–3XL') return ['M', 'L', 'XL', 'XXL', '3XL'];
    if (product.sizes === 'L–XXL') return ['L', 'XL', 'XXL'];
    if (product.sizes === 'Free size') return ['Free size'];
    return ['Unspecified'];
  };

  const colorOptions = (product: Product, size?: string | null): string[] => {
    const variants = product.variants ?? [];
    return [...new Set(
      variants
        .filter((variant) => !size || variant.size === size)
        .map((variant) => variant.color),
    )];
  };

  const findVariant = (product: Product, size: string, color: string) =>
    product.variants?.find((variant) => variant.size === size && variant.color === color);

  const departmentLabel = (item: Department) => {
    if (language === 'en') return item;
    return { All: t.all, Men: t.men, Women: t.women, Kids: t.kids }[item];
  };

  const toggleFavorite = (id: string) => {
    setFavorites((current) =>
      current.includes(id) ? current.filter((saved) => saved !== id) : [...current, id],
    );
  };

  const addToCart = (product: Product, size: string | null, color: string | null) => {
    if (!size || !color) {
      Alert.alert(t.chooseVariant);
      return false;
    }
    const variant = findVariant(product, size, color);
    if (!variant) {
      Alert.alert(t.chooseVariant);
      return false;
    }
    const itemId = `${product.id}::${variant.id}`;
    const existing = cartItems.find((item) => item.id === itemId);
    if (variant.stock <= (existing?.quantity ?? 0)) {
      Alert.alert(t.outOfStock, t.stockError);
      return false;
    }
    setCartItems((current) => {
      const existing = current.find((item) => item.id === itemId);
      if (existing) {
        return current.map((item) =>
          item.id === itemId ? { ...item, quantity: item.quantity + 1 } : item,
        );
      }
      return [...current, {
        id: itemId,
        productId: product.id,
        variantId: variant.id,
        size,
        color,
        quantity: 1,
      }];
    });
    return true;
  };

  const changeCartQuantity = (itemId: string, difference: number) => {
    const item = cartItems.find((candidate) => candidate.id === itemId);
    if (!item) return;
    const product = products.find((candidate) => candidate.id === item.productId);
    const variant = product?.variants?.find((candidate) => candidate.id === item.variantId);
    if (difference > 0 && variant && item.quantity >= variant.stock) {
      Alert.alert(t.outOfStock, t.stockError);
      return;
    }
    setCartItems((current) =>
      current.map((item) =>
        item.id === itemId
          ? { ...item, quantity: Math.max(1, item.quantity + difference) }
          : item,
      ),
    );
  };

  const adjustStock = (
    productId: string,
    variantId: string,
    difference: number,
    kind: StockMovement['kind'],
  ) => {
    const product = products.find((item) => item.id === productId);
    const variant = product?.variants?.find((item) => item.id === variantId);
    if (!variant) return;
    if (difference < 0 && variant.stock < Math.abs(difference)) {
      Alert.alert(t.outOfStock);
      return;
    }
    setProducts((current) =>
      current.map((item) =>
        item.id !== productId
          ? item
          : {
              ...item,
              variants: item.variants?.map((entry) =>
                entry.id === variantId
                  ? { ...entry, stock: Math.max(0, entry.stock + difference) }
                  : entry,
              ),
            },
      ),
    );
    setMovements((current) => [{
      id: `move-${Date.now()}-${Math.random().toString(36).slice(2, 7)}`,
      productId,
      variantId,
      kind,
      quantity: difference,
      createdAt: new Date().toISOString(),
    }, ...current].slice(0, 200));
  };

  const addDraftProduct = () => {
    const price = Number(newPrice);
    const sizeList = newSizes.split(',').map((item) => item.trim()).filter(Boolean);
    const colorList = newColors.split(',').map((item) => item.trim()).filter(Boolean);
    if (!newName.trim() || !newNameAm.trim() || !Number.isFinite(price) || price <= 0 || !sizeList.length || !colorList.length) {
      Alert.alert(t.addDraft, t.productFieldsRequired);
      return;
    }
    const id = `draft-${Date.now()}`;
    const variants = sizeList.flatMap((size, sizeIndex) =>
      colorList.map((color, colorIndex) => ({
        id: `${id}-v${sizeIndex}-${colorIndex}`,
        size,
        color,
        stock: 0,
      })),
    );
    const product: Product = {
      id,
      department: newDepartment,
      name: newName.trim(),
      nameAm: newNameAm.trim(),
      price,
      sizes: sizeList.join(', '),
      emoji: '🛍️',
      tint: '#E8E2D7',
      variants,
      status: 'draft',
      sourceUrl: draftSourceId ? telegramCandidates.find((item) => item.id === draftSourceId)?.postUrl : undefined,
      sourceChannel: draftSourceId ? telegramCandidates.find((item) => item.id === draftSourceId)?.channel : undefined,
      sourceDate: draftSourceId ? telegramCandidates.find((item) => item.id === draftSourceId)?.postedAt : undefined,
    };
    setProducts((current) => [product, ...current]);
    if (draftSourceId) updateTelegramCandidate(draftSourceId, 'drafted');
    setDraftSourceId(null);
    setNewName('');
    setNewNameAm('');
    setNewPrice('');
    Alert.alert(t.draft, t.verifyPublish);
  };

  const requestOrder = () => {
    if (!cartItems.length) return;
    if (!customerName.trim() || !customerPhone.trim()) {
      Alert.alert(t.contactRequired);
      return;
    }
    const order: StoreOrder = {
      id: `BL-${Date.now().toString().slice(-6)}`,
      customerName: customerName.trim(),
      customerPhone: customerPhone.trim(),
      fulfillment,
      items: cartItems.map((item) => ({ ...item })),
      total: cartSubtotal,
      status: 'requested',
      createdAt: new Date().toISOString(),
    };
    setOrders((current) => [order, ...current]);
    setCartItems([]);
    setCustomerName('');
    setCustomerPhone('');
    setShowOrderRequest(false);
    setShowCart(false);
    Alert.alert(order.id, t.orderSaved);
  };

  const updateOrderStatus = (orderId: string, nextStatus: OrderStatus) => {
    const order = orders.find((item) => item.id === orderId);
    if (!order) return;
    if (nextStatus === 'confirmed' && order.status === 'requested') {
      const enoughStock = order.items.every((line) => {
        const product = products.find((item) => item.id === line.productId);
        const variant = product?.variants?.find((item) => item.id === line.variantId);
        return Boolean(variant && variant.stock >= line.quantity);
      });
      if (!enoughStock) {
        Alert.alert(t.stockError);
        return;
      }
      setProducts((current) =>
        current.map((product) => ({
          ...product,
          variants: product.variants?.map((variant) => {
            const line = order.items.find(
              (item) => item.productId === product.id && item.variantId === variant.id,
            );
            return line ? { ...variant, stock: variant.stock - line.quantity } : variant;
          }),
        })),
      );
      setMovements((current) => [
        ...order.items.map((line) => ({
          id: `move-${Date.now()}-${line.id}`,
          productId: line.productId,
          variantId: line.variantId,
          kind: 'order' as const,
          quantity: -line.quantity,
          createdAt: new Date().toISOString(),
        })),
        ...current,
      ].slice(0, 200));
    } else if (nextStatus === 'cancelled' && order.status === 'confirmed') {
      setProducts((current) =>
        current.map((product) => ({
          ...product,
          variants: product.variants?.map((variant) => {
            const line = order.items.find(
              (item) => item.productId === product.id && item.variantId === variant.id,
            );
            return line ? { ...variant, stock: variant.stock + line.quantity } : variant;
          }),
        })),
      );
      setMovements((current) => [
        ...order.items.map((line) => ({
          id: `return-${Date.now()}-${line.id}`,
          productId: line.productId,
          variantId: line.variantId,
          kind: 'return' as const,
          quantity: line.quantity,
          createdAt: new Date().toISOString(),
        })),
        ...current,
      ].slice(0, 200));
    }
    setOrders((current) =>
      current.map((item) => item.id === orderId ? { ...item, status: nextStatus } : item),
    );
  };

  const removeCartItem = (itemId: string) => {
    setCartItems((current) => current.filter((item) => item.id !== itemId));
  };

  const departments: Department[] = ['All', 'Men', 'Women', 'Kids'];

  const openProductDetails = (product: Product) => {
    setSelectedProduct(product);
    setSelectedSize(null);
    setSelectedColor(null);
    setSelectedImageIndex(0);
  };
  const selectedVariant =
    selectedProduct && selectedSize && selectedColor
      ? findVariant(selectedProduct, selectedSize, selectedColor)
      : undefined;

  if (!storageReady) {
    return (
      <GestureHandlerRootView style={styles.gestureRoot}>
        <SafeAreaView style={styles.safe}>
          <Text style={styles.loadingText}>Belika…</Text>
        </SafeAreaView>
      </GestureHandlerRootView>
    );
  }

  return (
    <GestureHandlerRootView style={styles.gestureRoot}>
    <SafeAreaView style={styles.safe}>
      <StatusBar barStyle="dark-content" backgroundColor={COLORS.paper} />
      <View style={styles.header}>
        <View style={styles.brand}>
          <View style={styles.brandMark}><Text style={styles.brandMarkText}>B</Text></View>
          <View>
            <Text style={styles.brandName}>Belika</Text>
            <Text style={styles.brandSub}>MEN  ·  WOMEN  ·  KIDS</Text>
          </View>
        </View>
        <View style={styles.headerActions}>
          <TouchableOpacity style={styles.smallButton} onPress={() => setLanguage(language === 'en' ? 'am' : 'en')}>
            <Text style={styles.languageText}>{language === 'en' ? 'አማ' : 'EN'}</Text>
          </TouchableOpacity>
          <View style={styles.headerIconWrap}>
            <TouchableOpacity style={styles.iconButton} onPress={() => setShowWishlist(!showWishlist)}>
              <Text style={styles.iconText}>{showWishlist ? '♥' : '♡'}</Text>
            </TouchableOpacity>
            {favorites.length > 0 && (
              <View style={styles.cartBadge}>
                <Text style={styles.cartBadgeText}>{favorites.length}</Text>
              </View>
            )}
          </View>
          <TouchableOpacity
            style={styles.iconButton}
            onPress={() => setShowCart(true)}
          >
            <Text style={styles.iconText}>▱</Text>
            {cartCount > 0 && <View style={styles.cartBadge}><Text style={styles.cartBadgeText}>{cartCount}</Text></View>}
          </TouchableOpacity>
          <TouchableOpacity
            style={styles.iconButton}
            onPress={() => setShowStaff(true)}
            accessibilityLabel={t.staff}
          >
            <Text style={styles.iconText}>⚙</Text>
          </TouchableOpacity>
        </View>
      </View>

      <ScrollView contentContainerStyle={styles.content} keyboardShouldPersistTaps="handled">
        <View style={styles.searchBox}>
          <Text style={styles.searchIcon}>⌕</Text>
          <TextInput
            value={query}
            onChangeText={setQuery}
            placeholder={t.search}
            placeholderTextColor={COLORS.muted}
            style={styles.searchInput}
          />
        </View>

        <View style={styles.banner}>
          <Text style={styles.bannerTop}>{t.bannerTop}</Text>
          <Text style={styles.bannerTitle}>{t.bannerTitle}</Text>
          <Text style={styles.bannerText}>{t.bannerText}</Text>
          <TouchableOpacity
            style={styles.bannerButton}
            onPress={() => { setShowWishlist(false); setDepartment('All'); setQuery(''); }}
          >
            <Text style={styles.bannerButtonText}>{t.bannerButton}  →</Text>
          </TouchableOpacity>
          <View style={styles.bannerAccent} />
        </View>

        <Text style={styles.sectionTitle}>{t.departments}</Text>
        <ScrollView horizontal showsHorizontalScrollIndicator={false} style={styles.chipRow}>
          {departments.map((item) => (
            <TouchableOpacity
              key={item}
              onPress={() => { setDepartment(item); setShowWishlist(false); }}
              style={[styles.chip, department === item && !showWishlist && styles.chipActive]}
            >
              <Text style={[styles.chipText, department === item && !showWishlist && styles.chipTextActive]}>
                {departmentLabel(item)}
              </Text>
            </TouchableOpacity>
          ))}
        </ScrollView>

        <View style={styles.listHeading}>
          <Text style={styles.sectionTitle}>{showWishlist ? t.wishlist : t.featured}</Text>
          <Text style={styles.countText}>{results.length}</Text>
        </View>

        {results.length === 0 ? (
          <View style={styles.emptyBox}>
            <Text style={styles.emptyIcon}>{showWishlist ? '♡' : '⌕'}</Text>
            <Text style={styles.emptyText}>
              {department === 'Kids' && !showWishlist ? t.kidsSoon : t.empty}
            </Text>
          </View>
        ) : (
          <View style={styles.grid}>
            {results.map((product) => (
              <View key={product.id} style={styles.card}>
                <View style={[styles.productPhoto, { backgroundColor: product.tint }]}>
                  <TouchableOpacity
                    activeOpacity={0.85}
                    style={styles.photoTapArea}
                    onPress={() => openProductDetails(product)}
                  >
                    {product.images && product.images.length > 0 ? (
                      <Image source={product.images[0]} style={styles.productRealPhoto} resizeMode="cover" />
                    ) : (
                      <Text style={styles.productEmoji}>{product.emoji}</Text>
                    )}
                    {(!product.images || product.images.length === 0) && (
                      <Text style={styles.photoLabel}>{t.samplePhoto}</Text>
                    )}
                  </TouchableOpacity>
                  <TouchableOpacity style={styles.heartButton} onPress={() => toggleFavorite(product.id)}>
                    <Text style={styles.heartText}>{favorites.includes(product.id) ? '♥' : '♡'}</Text>
                  </TouchableOpacity>
                </View>
                <TouchableOpacity onPress={() => openProductDetails(product)}>
                  <Text style={styles.productName} numberOfLines={2}>
                    {language === 'en' ? product.name : product.nameAm}
                  </Text>
                  <Text style={styles.detailsLink}>{t.viewDetails} →</Text>
                </TouchableOpacity>
                <Text style={styles.sizeText}>{t.sizes}: {product.sizes}</Text>
                <View style={styles.cardBottom}>
                  <Text style={styles.price}>ETB {product.price.toLocaleString('en-US')}</Text>
                  <TouchableOpacity
                    style={styles.addButton}
                    onPress={() => openProductDetails(product)}
                  >
                    <Text style={styles.addButtonText}>+</Text>
                  </TouchableOpacity>
                </View>
              </View>
            ))}
          </View>
        )}

        <Text style={styles.demoNote}>{t.demo}</Text>
      </ScrollView>

      <Modal
        visible={selectedProduct !== null}
        animationType="slide"
        onRequestClose={() => setSelectedProduct(null)}
      >
        <GestureHandlerRootView style={styles.gestureRoot}>
        <SafeAreaView style={styles.modalSafe}>
          {selectedProduct && (
            <>
              <View style={styles.detailHeader}>
                <TouchableOpacity onPress={() => setSelectedProduct(null)} style={styles.backButton}>
                  <Text style={styles.backText}>‹</Text>
                </TouchableOpacity>
                <Text style={styles.detailHeaderTitle}>{t.details}</Text>
                <View style={styles.backButton} />
              </View>
              <ScrollView contentContainerStyle={styles.detailContent}>
                <View style={[styles.detailPhoto, { backgroundColor: selectedProduct.tint }]}>
                  {selectedProduct.images && selectedProduct.images.length > 0 ? (
                    <ZoomablePhoto
                      key={`${selectedProduct.id}-${selectedImageIndex}`}
                      source={selectedProduct.images[selectedImageIndex] ?? selectedProduct.images[0]}
                      resizeMode="contain"
                    />
                  ) : (
                    <>
                      <Text style={styles.detailEmoji}>{selectedProduct.emoji}</Text>
                      <Text style={styles.photoLabel}>{t.samplePhoto}</Text>
                    </>
                  )}
                  {selectedProduct.images && selectedProduct.images.length > 1 && (
                    <View style={styles.photoCounter}>
                      <Text style={styles.photoCounterText}>
                        {selectedImageIndex + 1} / {selectedProduct.images.length}
                      </Text>
                    </View>
                  )}
                </View>
                {selectedProduct.images && selectedProduct.images.length > 1 && (
                  <ScrollView horizontal showsHorizontalScrollIndicator={false} style={styles.galleryRow}>
                    {selectedProduct.images.map((image, index) => (
                      <TouchableOpacity
                        key={index}
                        onPress={() => setSelectedImageIndex(index)}
                        style={[styles.galleryThumb, selectedImageIndex === index && styles.galleryThumbActive]}
                      >
                        <Image source={image} style={styles.galleryThumbImage} resizeMode="cover" />
                        <Text style={styles.galleryCaption}>{selectedProduct.imageLabels?.[index] ?? ''}</Text>
                      </TouchableOpacity>
                    ))}
                  </ScrollView>
                )}
                <Text style={styles.detailName}>
                  {language === 'en' ? selectedProduct.name : selectedProduct.nameAm}
                </Text>
                <Text style={styles.detailPrice}>
                  ETB {selectedProduct.price.toLocaleString('en-US')}
                </Text>
                <Text style={styles.detailSizeLabel}>{t.chooseSize}</Text>
                <View style={styles.sizeRow}>
                  {sizeOptions(selectedProduct).map((size) => (
                    <TouchableOpacity
                      key={size}
                      onPress={() => {
                        setSelectedSize(size);
                        if (!colorOptions(selectedProduct, size).includes(selectedColor ?? '')) {
                          setSelectedColor(null);
                        }
                      }}
                      style={[styles.sizeChip, selectedSize === size && styles.sizeChipActive]}
                    >
                      <Text style={[styles.sizeChipText, selectedSize === size && styles.sizeChipTextActive]}>
                        {size}
                      </Text>
                    </TouchableOpacity>
                  ))}
                </View>
                <Text style={styles.detailSizeLabel}>{t.colors}</Text>
                <View style={styles.sizeRow}>
                  {colorOptions(selectedProduct, selectedSize).map((color) => (
                    <TouchableOpacity
                      key={color}
                      onPress={() => setSelectedColor(color)}
                      style={[styles.sizeChip, selectedColor === color && styles.sizeChipActive]}
                    >
                      <Text style={[styles.sizeChipText, selectedColor === color && styles.sizeChipTextActive]}>
                        {color}
                      </Text>
                    </TouchableOpacity>
                  ))}
                </View>
                <Text style={styles.detailNote}>
                  {selectedVariant
                    ? selectedVariant.stock > 0
                      ? `${selectedVariant.stock} ${t.remaining}`
                      : t.outOfStock
                    : t.chooseVariant}
                </Text>

                <Text style={styles.detailNote}>{t.sizeHelp}</Text>
                <Text style={styles.detailNote}>{t.photoHelp}</Text>

                <TouchableOpacity
                  style={styles.detailAddButton}
                  onPress={() => {
                    if (selectedProduct && addToCart(selectedProduct, selectedSize, selectedColor)) {
                      setSelectedProduct(null);
                    }
                  }}
                >
                  <Text style={styles.detailAddButtonText}>{t.addToCart}  +</Text>
                </TouchableOpacity>
              </ScrollView>
            </>
          )}
        </SafeAreaView>
        </GestureHandlerRootView>
      </Modal>

      <Modal
        visible={showCart}
        animationType="slide"
        onRequestClose={() => setShowCart(false)}
      >
        <SafeAreaView style={styles.cartModalSafe}>
          <View style={styles.cartHeader}>
            <TouchableOpacity onPress={() => setShowCart(false)} style={styles.backButton}>
              <Text style={styles.backText}>‹</Text>
            </TouchableOpacity>
            <Text style={styles.cartHeaderTitle}>{t.cartTitle}</Text>
            <View style={styles.backButton} />
          </View>

          {cartItems.length === 0 ? (
            <View style={styles.emptyCartBox}>
              <Text style={styles.emptyIcon}>▱</Text>
              <Text style={styles.emptyText}>{t.emptyCart}</Text>
            </View>
          ) : (
            <>
              <ScrollView contentContainerStyle={styles.cartContent}>
                {cartItems.map((item) => {
                  const product = products.find((candidate) => candidate.id === item.productId);
                  if (!product) return null;
                  return (
                    <View key={item.id} style={styles.cartItem}>
                      <View style={[styles.cartThumb, { backgroundColor: product.tint }]}>
                        {product.images && product.images.length > 0 ? (
                          <Image source={product.images[0]} style={styles.cartThumbImage} resizeMode="cover" />
                        ) : (
                          <Text style={styles.cartThumbEmoji}>{product.emoji}</Text>
                        )}
                      </View>
                      <View style={styles.cartItemInfo}>
                        <Text style={styles.cartItemName}>
                          {language === 'en' ? product.name : product.nameAm}
                        </Text>
                        <Text style={styles.cartItemMeta}>
                          {`${t.sizes}: ${item.size} · ${item.color}`}
                        </Text>
                        <Text style={styles.cartItemPrice}>
                          ETB {product.price.toLocaleString('en-US')}
                        </Text>
                        <View style={styles.cartControls}>
                          <TouchableOpacity
                            style={styles.quantityButton}
                            onPress={() => changeCartQuantity(item.id, -1)}
                          >
                            <Text style={styles.quantityButtonText}>−</Text>
                          </TouchableOpacity>
                          <Text style={styles.quantityText}>{item.quantity}</Text>
                          <TouchableOpacity
                            style={styles.quantityButton}
                            onPress={() => changeCartQuantity(item.id, 1)}
                          >
                            <Text style={styles.quantityButtonText}>+</Text>
                          </TouchableOpacity>
                          <TouchableOpacity
                            style={styles.removeButton}
                            onPress={() => removeCartItem(item.id)}
                          >
                            <Text style={styles.removeText}>{t.remove}</Text>
                          </TouchableOpacity>
                        </View>
                      </View>
                    </View>
                  );
                })}
              </ScrollView>
              <View style={styles.cartFooter}>
                <View style={styles.cartSubtotalRow}>
                  <Text style={styles.cartSubtotalLabel}>{t.subtotal}</Text>
                  <Text style={styles.cartSubtotalValue}>
                    ETB {cartSubtotal.toLocaleString('en-US')}
                  </Text>
                </View>
                <TouchableOpacity
                  style={styles.orderRequestButton}
                  onPress={() => setShowOrderRequest(true)}
                >

                </TouchableOpacity>
                <Text style={styles.checkoutNote}>{t.checkoutLater}</Text>
              </View>

          )}

        visible={showOrderRequest}
        animationType="slide"
        onRequestClose={() => setShowOrderRequest(false)}
      >
        <SafeAreaView style={styles.cartModalSafe}>
          <View style={styles.cartHeader}>
            <TouchableOpacity onPress={() => setShowOrderRequest(false)} style={styles.backButton}>
              <Text style={styles.backText}>‹</Text>
            </TouchableOpacity>
            <Text style={styles.cartHeaderTitle}>{t.orderRequest}</Text>
            <View style={styles.backButton} />
          </View>
          <ScrollView contentContainerStyle={styles.formContent} keyboardShouldPersistTaps="handled">
            <Text style={styles.localWarning}>{t.localOnly}</Text>
            <TextInput
              value={customerName}
              onChangeText={setCustomerName}
              placeholder={t.customerName}
              placeholderTextColor={COLORS.muted}
              style={styles.formInput}
            />
            <TextInput
              value={customerPhone}
              onChangeText={setCustomerPhone}
              placeholder={t.customerPhone}
              placeholderTextColor={COLORS.muted}
              keyboardType="phone-pad"
              style={styles.formInput}
            />
            <Text style={styles.formLabel}>{t.fulfillment}</Text>
            <View style={styles.sizeRow}>
              {(['pickup', 'delivery'] as const).map((choice) => (
                <TouchableOpacity
                  key={choice}
                  onPress={() => setFulfillment(choice)}
                  style={[styles.sizeChip, fulfillment === choice && styles.sizeChipActive]}
                >
                  <Text style={[styles.sizeChipText, fulfillment === choice && styles.sizeChipTextActive]}>
                    {choice === 'pickup' ? t.pickup : t.delivery}
                  </Text>
                </TouchableOpacity>
              ))}
            </View>
            <Text style={styles.cartSubtotalValue}>
              {t.total}: ETB {cartSubtotal.toLocaleString('en-US')}
            </Text>
            <TouchableOpacity style={styles.primaryButton} onPress={requestOrder}>
              <Text style={styles.primaryButtonText}>{t.submitRequest}</Text>
            </TouchableOpacity>
            <Text style={styles.checkoutNote}>{t.checkoutLater}</Text>
          </ScrollView>
        </SafeAreaView>
      </Modal>

        visible={showStaff}
        animationType="slide"
        onRequestClose={() => setShowStaff(false)}
      >
        <SafeAreaView style={styles.staffSafe}>
          <View style={styles.cartHeader}>
            <TouchableOpacity onPress={() => setShowStaff(false)} style={styles.backButton}>
              <Text style={styles.backText}>‹</Text>
            </TouchableOpacity>
            <Text style={styles.cartHeaderTitle}>{t.staffTitle}</Text>
            <View style={styles.backButton} />
          </View>
          <ScrollView ref={staffScrollRef} contentContainerStyle={styles.staffContent} keyboardShouldPersistTaps="handled">
            <Text style={styles.localWarning}>{t.localOnly}</Text>
            <Text style={styles.demoNote}>{t.sampleWarning}</Text>

            <Text style={styles.staffSectionTitle}>{t.orders}</Text>
            {orders.length === 0 ? (
              <Text style={styles.emptyText}>{t.noOrders}</Text>
            ) : (
              orders.map((order) => (
                <View key={order.id} style={styles.staffCard}>
                  <Text style={styles.staffCardTitle}>{order.id} · {order.status === 'requested'
                    ? t.requested
                    : order.status === 'confirmed'
                      ? t.confirmed
                      : order.status === 'completed'
                        ? t.completed
                        : t.cancelled}</Text>
                  <Text style={styles.staffMeta}>{order.customerName} · {order.customerPhone}</Text>
                  <Text style={styles.staffMeta}>
                    {order.fulfillment === 'pickup' ? t.pickup : t.delivery} · {order.items.reduce((sum, item) => sum + item.quantity, 0)} · ETB {order.total.toLocaleString('en-US')}
                  </Text>
                  {order.items.map((line) => {
                    const product = products.find((item) => item.id === line.productId);
                    return (
                      <Text key={`${order.id}-${line.id}`} style={styles.staffMeta}>
                        {product?.name ?? line.productId} · {line.size} / {line.color} × {line.quantity}
                      </Text>
                    );
                  })}
                  {order.status === 'requested' && (
                    <View style={styles.staffActions}>
                      <TouchableOpacity
                        style={styles.primarySmallButton}
                        onPress={() => updateOrderStatus(order.id, 'confirmed')}
                      >
                        <Text style={styles.primarySmallButtonText}>{t.confirmOrder}</Text>
                      </TouchableOpacity>
                      <TouchableOpacity
                        style={styles.secondarySmallButton}
                        onPress={() => updateOrderStatus(order.id, 'cancelled')}
                      >
                        <Text style={styles.secondarySmallButtonText}>{t.cancelOrder}</Text>
                      </TouchableOpacity>
                    </View>
                  )}
                  {order.status === 'confirmed' && (
                    <View style={styles.staffActions}>
                      <TouchableOpacity
                        style={styles.primarySmallButton}
                        onPress={() => updateOrderStatus(order.id, 'completed')}
                      >
                        <Text style={styles.primarySmallButtonText}>{t.completeOrder}</Text>
                      </TouchableOpacity>
                      <TouchableOpacity
                        style={styles.secondarySmallButton}
                        onPress={() => updateOrderStatus(order.id, 'cancelled')}
                      >
                        <Text style={styles.secondarySmallButtonText}>{t.cancelOrder}</Text>
                      </TouchableOpacity>
                    </View>
                  )}
                </View>
              ))
            )}

            <Text style={styles.staffSectionTitle}>{t.addDraft}</Text>
            <Text style={styles.staffMeta}>{t.addProductHelp}</Text>
            <TextInput value={newName} onChangeText={setNewName} placeholder={t.productName} placeholderTextColor={COLORS.muted} style={styles.formInput} />
            <TextInput value={newNameAm} onChangeText={setNewNameAm} placeholder={t.productNameAm} placeholderTextColor={COLORS.muted} style={styles.formInput} />
            <TextInput value={newPrice} onChangeText={setNewPrice} placeholder={t.priceETB} placeholderTextColor={COLORS.muted} keyboardType="numeric" style={styles.formInput} />
            <Text style={styles.formLabel}>{t.department}</Text>
            <View style={styles.sizeRow}>
              {(['Men', 'Women', 'Kids'] as const).map((choice) => (
                <TouchableOpacity
                  key={choice}
                  onPress={() => setNewDepartment(choice)}
                  style={[styles.sizeChip, newDepartment === choice && styles.sizeChipActive]}
                >
                  <Text style={[styles.sizeChipText, newDepartment === choice && styles.sizeChipTextActive]}>
                    {departmentLabel(choice)}
                  </Text>
                </TouchableOpacity>
              ))}
            </View>
            <TextInput value={newSizes} onChangeText={setNewSizes} placeholder={t.sizesInput} placeholderTextColor={COLORS.muted} style={styles.formInput} />
            <TextInput value={newColors} onChangeText={setNewColors} placeholder={t.colorsInput} placeholderTextColor={COLORS.muted} style={styles.formInput} />
            <TouchableOpacity style={styles.primaryButton} onPress={addDraftProduct}>
              <Text style={styles.primaryButtonText}>{t.addDraft}</Text>
            </TouchableOpacity>

            <Text style={styles.staffSectionTitle}>{language === 'en' ? 'Telegram product review' : 'የቴሌግራም ምርት ግምገማ'}</Text>
            <Text style={styles.localWarning}>
              {language === 'en'
                ? `Preview posts in the requested 2-month window · ${pendingTelegramCount} still need review. This is a partial public-preview sample, not a full Telegram archive. Photos stay in Telegram: open the source posts below to compare them. No post changes stock automatically.`
                : `በ2 ወር የሙከራ ጊዜ የታዩ ፖስቶች · ${pendingTelegramCount} ገና ይገመገማሉ። ይህ ከህዝብ ቅድመ-እይታ የተወሰደ ከፊል ናሙና ነው። ፎቶዎች በቴሌግራም ይቆያሉ፤ ሁለቱን ፖስቶች ከዚህ በታች ከፍተው ያነጻጽሩ። ፖስት በራሱ ክምችት አይቀይርም።`}
            </Text>
            <View style={styles.sizeRow}>
              {(['pending', 'matches', 'restocks', 'all'] as const).map((filter) => (
                <TouchableOpacity key={filter} onPress={() => setCandidateFilter(filter)} style={[styles.sizeChip, candidateFilter === filter && styles.sizeChipActive]}>
                  <Text style={[styles.sizeChipText, candidateFilter === filter && styles.sizeChipTextActive]}>
                    {filter === 'pending' ? (language === 'en' ? `Pending ${pendingTelegramCount}` : `ያልተገመገመ ${pendingTelegramCount}`)
                      : filter === 'matches' ? (language === 'en' ? 'Possible duplicates' : 'ሊደጋገሙ የሚችሉ')
                        : filter === 'restocks' ? (language === 'en' ? 'Restocks' : 'ዳግም ክምችት')
                          : (language === 'en' ? 'All' : 'ሁሉም')}
                  </Text>
                </TouchableOpacity>
              ))}
            </View>
            {filteredTelegramCandidates.map((candidate) => {
              const related = candidate.groupId
                ? telegramCandidates.filter((item) => item.groupId === candidate.groupId && item.id !== candidate.id)
                : [];
              const status = candidate.status ?? 'pending';
              const kindLabel = candidate.kind === 'restock'
                ? (language === 'en' ? 'RESTOCK EVENT · quantity not given' : 'ዳግም ክምችት · መጠኑ አልተገለጸም')
                : candidate.kind === 'possible-duplicate'
                  ? (language === 'en' ? 'POSSIBLE MATCH · compare photos' : 'ሊደጋገም ይችላል · ፎቶዎቹን ያነጻጽሩ')

                    : status === 'restock-reviewed' ? (language === 'en' ? 'Restock noted · stock unchanged' : 'ዳግም ክምችት ተመልክቷል · ክምችት አልተቀየረም')
                      : status === 'ignored' ? (language === 'en' ? 'Ignored' : 'ተወግዷል')
                        : status === 'prepared' ? (language === 'en' ? 'Draft form prepared' : 'የረቂቅ ቅጽ ተዘጋጅቷል')
                          : (language === 'en' ? 'Needs review' : 'ግምገማ ያስፈልጋል');
              return (
                <View key={candidate.id} style={styles.staffCard}>
                  <Text style={styles.candidateTag}>{kindLabel}</Text>
                  <Text style={styles.staffCardTitle}>{candidate.name} · ETB {candidate.price.toLocaleString('en-US')}</Text>
                  <Text style={styles.staffMeta}>{candidate.department} · {candidate.sizeSpec} · {candidate.postedAt} · {candidate.channel}</Text>
                  <Text style={styles.candidateStatus}>{statusLabel}</Text>
                  {candidate.note ? <Text style={styles.staffMeta}>{candidate.note}</Text> : null}
                  <View style={styles.staffActions}>
                    <TouchableOpacity style={styles.primarySmallButton} onPress={() => void openTelegramPost(candidate.postUrl)}>
                      <Text style={styles.primarySmallButtonText}>{language === 'en' ? 'Open original photo' : 'የመጀመሪያ ፎቶ ክፈት'}</Text>
                    </TouchableOpacity>
                    {candidate.kind !== 'restock' && status !== 'drafted' && (
                      <TouchableOpacity style={styles.secondarySmallButton} onPress={() => prepareTelegramDraft(candidate)}>
                        <Text style={styles.secondarySmallButtonText}>{language === 'en' ? 'Prepare draft' : 'ረቂቅ አዘጋጅ'}</Text>
                      </TouchableOpacity>
                    )}
                  </View>
                  {related.length > 0 && (
                    <View style={styles.compareBox}>
                      <Text style={styles.staffMeta}>{language === 'en' ? 'Compare related posts visually in Telegram:' : 'ተዛማጅ ፖስቶችን በቴሌግራም ያነጻጽሩ:'}</Text>
                      {related.map((other) => (
                        <TouchableOpacity key={other.id} style={styles.compareLink} onPress={() => void openTelegramPost(other.postUrl)}>

                        </TouchableOpacity>
                      ))}
                    </View>
                  )}
                  <View style={styles.staffActions}>
                    {candidate.kind === 'possible-duplicate' && candidate.groupId && (
                      <>
                        <TouchableOpacity style={styles.secondarySmallButton} onPress={() => updateTelegramGroup(candidate.groupId!, 'duplicate')}>
                          <Text style={styles.secondarySmallButtonText}>{language === 'en' ? 'Mark group duplicate' : 'ቡድኑን ድግግሞሽ በል'}</Text>
                        </TouchableOpacity>
                        <TouchableOpacity style={styles.secondarySmallButton} onPress={() => updateTelegramGroup(candidate.groupId!, 'kept')}>
                          <Text style={styles.secondarySmallButtonText}>{language === 'en' ? 'Keep as separate items' : 'እንደ ተለያዩ ያቆዩ'}</Text>
                        </TouchableOpacity>
                      </>
                    )}
                    {candidate.kind === 'restock' && status !== 'restock-reviewed' && (
                      <TouchableOpacity style={styles.secondarySmallButton} onPress={() => updateTelegramCandidate(candidate.id, 'restock-reviewed')}>
                        <Text style={styles.secondarySmallButtonText}>{language === 'en' ? 'Note restock (no qty)' : 'ዳግም ክምችት ምልክት አድርግ'}</Text>
                      </TouchableOpacity>
                    )}
                    {candidate.kind === 'product' && status !== 'ignored' && status !== 'drafted' && (
                      <TouchableOpacity style={styles.secondarySmallButton} onPress={() => updateTelegramCandidate(candidate.id, 'ignored')}>
                        <Text style={styles.secondarySmallButtonText}>{language === 'en' ? 'Ignore post' : 'ፖስቱን ተወው'}</Text>
                      </TouchableOpacity>
                    )}
                  </View>
                </View>
              );
            })}

            <Text style={styles.staffSectionTitle}>{t.stock}</Text>
            {products.map((product) => (
              <View key={product.id} style={styles.staffCard}>
                <View style={styles.staffProductHeading}>
                  <View style={{ flex: 1 }}>
                    <Text style={styles.staffCardTitle}>
                      {language === 'en' ? product.name : product.nameAm}
                    </Text>
                    <Text style={styles.staffMeta}>
                      {product.department} · {product.status === 'draft' ? t.draft : t.published}
                    </Text>
                  </View>
                  {product.status === 'draft' && (
                    <TouchableOpacity
                      style={styles.primarySmallButton}
                      onPress={() => setProducts((current) => current.map((item) =>
                        item.id === product.id ? { ...item, status: 'published' } : item,
                      ))}
                    >
                      <Text style={styles.primarySmallButtonText}>{t.verifyPublish}</Text>
                    </TouchableOpacity>
                  )}
                </View>
                {(product.variants ?? []).map((variant) => (
                  <View key={variant.id} style={styles.stockRow}>
                    <View style={styles.stockTextWrap}>
                      <Text style={styles.staffMeta}>{variant.size} · {variant.color}</Text>
                      <Text style={styles.stockNumber}>{t.stock}: {variant.stock}</Text>
                    </View>
                    <TouchableOpacity
                      style={styles.stockAdjustButton}
                      onPress={() => adjustStock(product.id, variant.id, 1, 'restock')}
                    >
                      <Text style={styles.stockAdjustText}>+1</Text>
                    </TouchableOpacity>
                    <TouchableOpacity
                      style={styles.saleAdjustButton}
                      onPress={() => adjustStock(product.id, variant.id, -1, 'sale')}
                    >
                      <Text style={styles.saleAdjustText}>−1</Text>
                    </TouchableOpacity>
                  </View>
                ))}
              </View>
            ))}

            <Text style={styles.staffSectionTitle}>{t.inventoryHistory}</Text>
            {movements.length === 0 ? (
              <Text style={styles.emptyText}>{t.noMovements}</Text>
            ) : (
              movements.slice(0, 20).map((movement) => {
                const product = products.find((item) => item.id === movement.productId);
                const variant = product?.variants?.find((item) => item.id === movement.variantId);
                return (
                  <Text key={movement.id} style={styles.staffMeta}>
                    {new Date(movement.createdAt).toLocaleString()} · {product?.name ?? movement.productId} · {variant?.size ?? ''} {variant?.color ?? ''} · {movement.kind} {movement.quantity > 0 ? '+' : ''}{movement.quantity}
                  </Text>
                );
              })
            )}
          </ScrollView>
        </SafeAreaView>
      </Modal>
    </SafeAreaView>
    </GestureHandlerRootView>
  );
}
